
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
import json, os, shlex, subprocess, tempfile
root=Path('/Users/thomas/Projects/skills')
out=root/'docs/evaluation/editorial-329/followup/validation/second'
cleanup='/Users/thomas/.agents/skills/kntnt/features/session-cleanup/scripts/session_cleanup.py'
def register(kind, value):
 subprocess.run(['uv','run',cleanup,'add',kind,str(value),'Editorial source-fidelity final four checks'],cwd=root,check=True,capture_output=True)
register('pid',os.getpid())
temp=Path(tempfile.mkdtemp(prefix='kntnt-fidelity-final-tests-')).resolve();register('path',temp)
(out/'scratch.json').write_text(json.dumps({'path':str(temp),'pid':os.getpid(),'removed':False},indent=2)+'\n')
prior=json.loads((out.parent/'test-results.json').read_text())
env=dict(os.environ);env['PYTEST_ADDOPTS']='--basetemp='+str(temp/'pytest')
def check(row):
 with (out/(row['check']+'.log')).open('w') as log:
  result=subprocess.run(shlex.split(row['command']),cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT)
 return {**row,'exit':result.returncode}
with ThreadPoolExecutor(max_workers=4) as pool:
 results=list(pool.map(check,prior['results']))
(out/'test-results.json').write_text(json.dumps({'revision':'82db439','pytest_basetemp':str(temp/'pytest'),'results':results},indent=2)+'\n')
print(json.dumps(results),flush=True)
