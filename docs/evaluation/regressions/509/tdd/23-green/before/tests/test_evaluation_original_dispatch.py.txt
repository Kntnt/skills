def source(case: dict[str, Any], kind: str) -> bytes:
    """Read lossless copies against independent original byte identities."""
    identity = case["files"][kind]
    data = gzip.decompress((EVIDENCE / identity["fixture"]).read_bytes())
    assert len(data) == identity["bytes"]
    assert hashlib.sha256(data).hexdigest() == identity["sha256"]
    return data


def recorder(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
    case: dict[str, Any],
    **changes: Any,
) -> argparse.Namespace:
    """Supply existing opaque transport at the external subprocess seam.

    No native session is launched. Registration and termination are external
    lifecycle operations replaced by inert stand-ins for this offline test.
    """
    auth = tmp_path / "auth"
    auth.mkdir()
    (auth / "hooks.json").write_text('{"hooks":{}}')
    (auth / "auth.json").write_text("synthetic credential")
    native = tmp_path / "native"
    native.write_bytes(b"synthetic executable, never executed")
    packet = tmp_path / "packet"
    args = argparse.Namespace(
        packet=packet,
        cwd=tmp_path,
        model=case["model"],
        effort=case["effort"],
        version=case["version"],
        parent="synthetic-parent",
        role="neutral",
        timeout=1200,
        native=native,
        auth_home=auth,
        cleanup_script=tmp_path / "cleanup",
        inventory_root=[],
        source_contract="original-completed-v1",
    )
    args.observed = {}
    monkeypatch.setattr(DISPATCH.os, "getpgrp", os.getpid)
    monkeypatch.setattr(DISPATCH, "register", lambda *args: None)
    monkeypatch.setattr(DISPATCH, "stop_owned", lambda *unused: None)
    monkeypatch.setattr(
        DISPATCH.subprocess,
        "check_output",
        lambda *args, **kwargs: "codex-cli 0.160.0\n",
    )

    class Transport:
        """Finish the exact stored source; mutations are explicitly synthetic."""

        pid = os.getpid()
        returncode = 0

        def __init__(self, command: list[str], **kwargs: Any) -> None:
            self.options = kwargs
            args.observed.update(command=command, env=kwargs["env"])

        def communicate(self, prompt: bytes, timeout: float) -> None:
            assert prompt == changes.get("prompt", source(case, "prompt"))
            if changes.get("timeout"):
                raise subprocess.TimeoutExpired("synthetic native", timeout)
            sessions = Path(self.options["env"]["CODEX_HOME"]) / "sessions"
            sessions.mkdir()
            (sessions / "original.jsonl").write_bytes(
                changes.get("raw", source(case, "raw"))
            )
            self.options["stdout"].write(changes.get("events", source(case, "events")))
            (packet / "terminal.txt").write_bytes(
                changes.get("terminal", source(case, "terminal"))
            )
            self.returncode = changes.get("exit", 0)
            args.observed["completed"] = True

    monkeypatch.setattr(DISPATCH.subprocess, "Popen", Transport)
    monkeypatch.setattr(
        DISPATCH.subprocess,
        "run",
        lambda *unused, **options: SimpleNamespace(stdout="synthetic start identity"),
    )
    monkeypatch.setattr(DISPATCH.os, "getpgid", lambda pid: pid)
    return args


def test_selected_original_contract_acquires_false_retention(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    """The real recorder's default guard must not decide the explicit contract."""
    case = CASES[0]
    args = recorder(tmp_path, monkeypatch, case)
    result = DISPATCH.run(args, source(case, "prompt"))
    assert hashlib.sha256(result).hexdigest() == case["files"]["terminal"]["sha256"]
    receipt = json.loads((args.packet / "receipt.json").read_text())
    assert receipt["binding"]["source_contract"] == "original-completed-v1"
    assert receipt["binding"]["final_retained_source"]["complete"] is False
    assert not (args.packet / "runtime").exists()


@pytest.mark.parametrize("case", CASES, ids=[case["case"] for case in CASES])
def test_all_original_sources_are_acquired_without_rewriting_retention(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch, case: dict[str, Any]
) -> None:
    """Stored originals bind even where per-message retention is incomplete."""
    args = recorder(tmp_path, monkeypatch, case)
    result = DISPATCH.run(args, source(case, "prompt"))
    assert hashlib.sha256(result).hexdigest() == case["files"]["terminal"]["sha256"]
    rows = [json.loads(line) for line in source(case, "raw").splitlines()]
    original = copy.deepcopy(rows)
    binding, bound = DISPATCH.bind_original_trace(
        rows, source(case, "prompt"), case["model"], case["effort"], case["version"]
    )
    assert bound == result
    assert rows == original
    assert binding["native_input_id"] != binding["original_input_id"]
    if case["case"] == "A":
        assert binding["initial_retained_source"]["complete"] is False
        assert binding["final_retained_source"]["complete"] is True
    else:
        assert binding["initial_retained_source"]["complete"] is True
        assert binding["final_retained_source"]["complete"] is False
    with pytest.raises(DISPATCH.CaptureError):
        DISPATCH.bind_trace(
            rows, source(case, "prompt"), case["model"], case["effort"], case["version"]
        )

    )

@pytest.mark.parametrize(
    "mutation",
    [
        "foreign-thread",
        "missing-start",
        "duplicate-start",
        "missing-completion",
        "duplicate-completion",
        "failed-turn",
        "altered-final",
        "missing-final",
        "duplicate-item",
    ],
)
def test_synthetic_exec_transport_conflicts_fail(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch, mutation: str
) -> None:
    """Native raw binding also requires an agreeing successful exec transport."""
    case = CASES[0]
    events = [json.loads(line) for line in source(case, "events").splitlines()]
    finals = [
        i
        for i, event in enumerate(events)
        if event.get("type") == "item.completed"
        and event.get("item", {}).get("type") == "agent_message"
    ]
    if mutation == "foreign-thread":
        events[0]["thread_id"] = "foreign"
    elif mutation == "missing-start":
        events = [e for e in events if e["type"] != "turn.started"]
    elif mutation == "duplicate-start":
        events.insert(2, {"type": "turn.started"})
    elif mutation == "missing-completion":
        events = [e for e in events if e["type"] != "turn.completed"]
    elif mutation == "duplicate-completion":
        events.append(copy.deepcopy(events[-1]))
    elif mutation == "failed-turn":
        events.append(
            {"type": "turn.failed", "error": {"message": "synthetic failure"}}
        )
    elif mutation == "altered-final":
        events[finals[-1]]["item"]["text"] += "synthetic mismatch"
    elif mutation == "missing-final":
        events.pop(finals[-1])
    else:
        events.insert(finals[-1] + 1, copy.deepcopy(events[finals[-1]]))
    data = b"".join(json.dumps(e).encode() + b"\n" for e in events)
    args = recorder(tmp_path, monkeypatch, case, events=data)
    with pytest.raises(DISPATCH.CaptureError, match="exec"):
        DISPATCH.run(args, source(case, "prompt"))
    assert not (args.packet / "result.txt").exists()
    assert json.loads((args.packet / "receipt.json").read_text())["status"] == "failed"

@pytest.mark.parametrize("boundary", ["reused-pid", "unknown-start", "foreign-group"])
def test_recorder_refuses_to_stop_unverified_identity(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch, boundary: str
) -> None:
    """A completed leader's reused identity never authorizes a group signal."""
    actual_stop = DISPATCH.stop_owned
    case = CASES[0]
    args = recorder(tmp_path, monkeypatch, case)
    signals: list[tuple[int, int]] = []

    def signal_group(pid: int, sig: int) -> None:
        signals.append((pid, sig))
        raise ProcessLookupError

    def start(*unused: Any, **options: Any) -> SimpleNamespace:
        value = (
            "different start"
            if args.observed.get("completed") and boundary == "reused-pid"
            else "synthetic start identity"
        )
        return SimpleNamespace(stdout="" if boundary == "unknown-start" else value)

    monkeypatch.setattr(DISPATCH, "stop_owned", actual_stop)
    monkeypatch.setattr(DISPATCH.subprocess, "run", start)
    monkeypatch.setattr(DISPATCH.os, "killpg", signal_group)
    if boundary == "foreign-group":
        monkeypatch.setattr(DISPATCH.os, "getpgid", lambda pid: pid + 1)
    with pytest.raises(DISPATCH.CaptureError, match="ownership"):
        DISPATCH.run(args, source(case, "prompt"))
    assert signals == []
    assert not (args.packet / "result.txt").exists()

        assert observed["retained_context"]["user_messages"][0]["complete"] is False
        assert observed["retained_context"]["user_messages"][0]["utf8_bytes"] == 3600
        assert (
            observed["retained_context"]["user_messages"][0]["utf8_bytes"]
