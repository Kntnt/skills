This evaluator helper does not change a shipped Skill. A successful receipt
establishes this instrumented boundary, never an opaque native spawn's input.
"""

class CaptureError(Exception):
    """The authentic native records do not establish a complete dispatch."""


def digest(data: bytes) -> str:
    """Bind evidence without changing its bytes."""
    return hashlib.sha256(data).hexdigest()


def write_json(path: Path, value: Any) -> None:
    """Write recorder metadata separately from native receipts."""
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def message_bytes(payload: dict[str, Any]) -> bytes:
    """Read only complete plaintext native message blocks."""

    # Reject unavailable or non-text message content.
    blocks = payload.get("content", [])
    if not blocks or any(
        block.get("type") not in ("input_text", "output_text", "text")
        or not isinstance(block.get("text"), str)
        for block in blocks
    ):
        raise CaptureError("message contains unavailable or non-text content")

    return "".join(block["text"] for block in blocks).encode("utf-8")


def retained_message(row: dict[str, Any], turn_id: str, role: str) -> bytes:
    """Require complete retained content and agreeing message/turn identities."""

    # Reject incomplete or conflicting retained identity before reading bytes.
    payload = row["payload"]
    identity = payload.get("id")
    retained = row.get("metadata", {}).get("retained_source", {})
    if (
        not isinstance(identity, str)
        or not identity
        or payload.get("internal_chat_message_metadata_passthrough", {}).get("turn_id")
        != turn_id
        or retained.get("complete") is not True
        or retained.get("id")
        != {"message_id": identity, "turn_id": turn_id, "role": role}
    ):
        raise CaptureError("missing or conflicting retained message identity")

    return message_bytes(payload)


def native_item_bytes(payload: dict[str, Any], thread_id: str, turn_id: str) -> bytes:
    """Read the native v2 completed item independently of its retained copy."""

    # Reject completed items with foreign or unavailable identity.
    item = payload["item"]
    expected_type = "text" if item["type"] == "UserMessage" else "Text"
    if payload.get("thread_id") != thread_id or payload.get("turn_id") != turn_id:
        raise CaptureError("native completed item identity disagrees")
    if not isinstance(item.get("id"), str) or not item["id"]:
        raise CaptureError("native completed item identity unavailable")

    # Read only complete plaintext in the observed native item representation.
    blocks = item.get("content", [])
    if not blocks or any(
        block.get("type") != expected_type or not isinstance(block.get("text"), str)
        for block in blocks
    ):
        raise CaptureError("native completed item plaintext unavailable")

    return "".join(block["text"] for block in blocks).encode("utf-8")


def bind_trace(
    records: list[dict[str, Any]], prompt: bytes, model: str, effort: str, version: str
) -> tuple[dict[str, Any], bytes]:
    """Bind one initial v2 task and both final representations in a fresh turn."""

    # Reject absent or ambiguous native session/context boundaries.
    metas = [row["payload"] for row in records if row.get("type") == "session_meta"]
    contexts = [
        (ordinal, row["payload"])
        for ordinal, row in enumerate(records, 1)
        if row.get("type") == "turn_context"
    ]
    if len(metas) != 1 or len(contexts) != 1:
        raise CaptureError("missing or ambiguous native thread/context")

    # Reject inherited or contradictory native ancestry.
    meta = metas[0]
    thread_id = meta.get("id")
    if (
        not isinstance(thread_id, str)
        or not thread_id
        or meta.get("session_id") != thread_id
    ):
        raise CaptureError("missing or conflicting native thread identity")
    if (
        meta.get("parent_thread_id") is not None
        or meta.get("source") != "exec"
        or meta.get("thread_source") != "user"
    ):
        raise CaptureError("native session is not a fresh top-level exec")

    # Bind the installed native version and actual effective turn seat.
    if meta.get("cli_version") != version:
        raise CaptureError("actual native version differs from freeze")
    context_ordinal, context = contexts[0]
    turn_id = context.get("turn_id")
    if (
        not isinstance(turn_id, str)
        or not turn_id
        or context.get("root_turn_id") != turn_id
    ):
        raise CaptureError("missing or conflicting native turn identity")
    if context.get("model") != model or context.get("effort") != effort:
        raise CaptureError("actual native model/effort differs from freeze")

    # Select the initial task boundary, never a later matching user echo.
    inputs: list[tuple[int, dict[str, Any]]] = []
    finals: list[tuple[int, dict[str, Any]]] = []
    native_inputs: list[tuple[int, dict[str, Any]]] = []
    native_finals: list[tuple[int, dict[str, Any]]] = []
    starts: list[tuple[int, dict[str, Any]]] = []
    completions: list[tuple[int, dict[str, Any]]] = []
    for ordinal, row in enumerate(records, 1):
        # Collect retained message boundaries, including marked earlier tasks.
        payload = row.get("payload", {})
        if row.get("type") == "response_item" and payload.get("type") == "message":
            kinds = payload.get("internal_chat_message_metadata_passthrough", {}).get(
                "content_item_kinds", []
            )
            if payload.get("role") == "user" and (
                ordinal > context_ordinal
                or "user_input_order" in row.get("metadata", {})
                or "user.text" in kinds
            ):
                inputs.append((ordinal, row))
            if (
                payload.get("role") == "assistant"
                and payload.get("phase") == "final_answer"
            ):
                finals.append((ordinal, row))

        # Collect native lifecycle and completed boundaries with their identity.
        if row.get("type") == "event_msg":
            if payload.get("type") == "task_started":
                starts.append((ordinal, payload))
            if payload.get("type") == "task_complete":
                completions.append((ordinal, payload))

            # Reject foreign completed items before retaining their boundary.
            if payload.get("type") == "item_completed":
                if (
                    payload.get("thread_id") != thread_id
                    or payload.get("turn_id") != turn_id
                ):
                    raise CaptureError("native completed item identity disagrees")
                item = payload.get("item", {})
                if item.get("type") == "UserMessage":
                    native_inputs.append((ordinal, payload))
                if (
                    item.get("type") == "AgentMessage"
                    and item.get("phase") == "final_answer"
                ):
                    native_finals.append((ordinal, payload))

    # Reject incomplete or duplicate representations before pairing them.
    if any(
        len(items) != 1
        for items in (inputs, finals, native_inputs, native_finals, starts, completions)
    ):
        raise CaptureError("missing or ambiguous initial input/final/native completion")

    # Pair the unique retained and independently completed boundaries.
    input_ordinal, input_row = inputs[0]
    native_input_ordinal, native_input = native_inputs[0]
    final_ordinal, final_row = finals[0]
    native_final_ordinal, native_final = native_finals[0]
    start_ordinal, start = starts[0]
    completion_ordinal, completion = completions[0]

    # Reject altered or displaced initial user input.
    if (
        input_row.get("metadata", {}).get("user_input_order") != 0
        or input_row["payload"]
        .get("internal_chat_message_metadata_passthrough", {})
        .get("content_item_kinds")
        != ["user.text"]
        or retained_message(input_row, turn_id, "user") != prompt
        or native_item_bytes(native_input, thread_id, turn_id) != prompt
    ):
        raise CaptureError("authentic initial user content or order disagrees")

    # Reject conflicting final bytes, identity or completed-turn evidence.
    result = retained_message(final_row, turn_id, "assistant")
    if (
        native_final["item"].get("id") != final_row["payload"]["id"]
        or native_item_bytes(native_final, thread_id, turn_id) != result
        or completion.get("turn_id") != turn_id
        or completion.get("last_agent_message", "").encode("utf-8") != result
        or start.get("turn_id") != turn_id
        or start.get("root_turn_id") != turn_id
    ):
        raise CaptureError("native terminal content or turn identity disagrees")

    # Reject reordered boundaries that cannot bind the initial task to output.
    if (
        not start_ordinal
        < context_ordinal
        < input_ordinal
        < native_input_ordinal
        < native_final_ordinal
        < final_ordinal
        < completion_ordinal
    ):
        raise CaptureError("native record order does not bind initial input to result")

    # Return verifiable identities and exact byte digests.
    return {
        "thread_id": thread_id,
        "turn_id": turn_id,
        "input_ordinal": input_ordinal,
        "native_input_ordinal": native_input_ordinal,
        "final_ordinal": final_ordinal,
        "native_final_ordinal": native_final_ordinal,
        "completion_ordinal": completion_ordinal,
        "input_sha256": digest(prompt),
        "result_sha256": digest(result),
        "model": model,
        "effort": effort,
        "cli_version": version,
    }, result


def original_message(row: dict[str, Any], turn_id: str, role: str) -> bytes:
    """Require original identity while preserving an observed retention flag."""

    # A false flag limits retention, never removes original identity checks.
    payload = row["payload"]
    identity = payload.get("id")
    retained = row.get("metadata", {}).get("retained_source", {})
    if (
        not isinstance(identity, str)
        or not identity
        or payload.get("internal_chat_message_metadata_passthrough", {}).get("turn_id")
        != turn_id
        or not isinstance(retained.get("complete"), bool)
        or retained.get("id")
        != {"message_id": identity, "turn_id": turn_id, "role": role}
    ):
        raise CaptureError("missing or conflicting original message identity")

    return message_bytes(payload)


def corroborate_compaction(
    preceding: list[dict[str, Any]],
    compacted: dict[str, Any],
    input_row: dict[str, Any],
    prompt: bytes,
) -> None:
    """Check replacement identities without recovering any original from them."""

    # The separately authenticated order-zero original must still corroborate.
    history_inputs = [
        {"payload": payload, "metadata": metadata}
        for payload, metadata in zip(
            compacted["replacement_history"],
            compacted["replacement_history_metadata"],
            strict=True,
        )
        if payload.get("role") == "user"
        and (
            "user_input_order" in metadata
            or "user.text"
            in payload.get("internal_chat_message_metadata_passthrough", {}).get(
                "content_item_kinds", []
            )
        )
    ]
    if len(history_inputs) != 1 or history_inputs[0] != {
        "payload": input_row["payload"],
        "metadata": input_row["metadata"],
    }:
        raise CaptureError("compaction replacement initial source disagrees")

    # Each retained message refers to an earlier original revision and order.
    originals = {
        row["payload"]["id"]: row
        for row in preceding
        if row.get("type") == "response_item"
        and row.get("payload", {}).get("type") == "message"
        and "retained_source" in row.get("metadata", {})
    }
    retained = compacted["retained_context"]
    orders: list[int] = []
    identities: set[str] = set()
    for role in ("user", "assistant"):
        messages = retained.get(f"{role}_messages")
        if not isinstance(messages, list):
            raise CaptureError("compaction retained message source unavailable")
        for message in messages:
            original = originals.get(message.get("message_id"))
            if original is None:
                raise CaptureError("compaction retained message has no original")
            metadata = original["metadata"]
            identity = metadata["retained_source"]
            order = message.get("order")
            if (
                message.get("message_id") in identities
                or type(order) is not int
                or message.get("turn_id") != identity["id"]["turn_id"]
                or identity["id"]["role"] != role
                or message.get("revision") != identity.get("revision")
                or order != metadata.get("user_input_order")
                or type(message.get("complete")) is not bool
                or message["complete"] != identity["complete"]
                or not isinstance(message.get("text"), str)
                or (
                    message["complete"]
                    and message["text"].encode("utf-8")
                    != message_bytes(original["payload"])
                )
            ):
                raise CaptureError("compaction retained identity or content disagrees")
            identities.add(message["message_id"])
            orders.append(order)
    if (
        sorted(orders) != list(range(len(orders)))
        or retained.get("next_order") != len(orders)
        or input_row["payload"]["id"] not in identities
    ):
        raise CaptureError("compaction retained message order disagrees")


def bind_original_trace(
    records: list[dict[str, Any]], prompt: bytes, model: str, effort: str, version: str
) -> tuple[dict[str, Any], bytes]:
    """Fail closed on malformed untrusted original-source representations."""

    # Normalize malformed decoded shapes and invalid UTF-8 at this seam.
    try:
        return _bind_original_trace(records, prompt, model, effort, version)
    except (KeyError, TypeError, AttributeError, ValueError) as exc:
        raise CaptureError("malformed original source representation") from exc


def _bind_original_trace(
    records: list[dict[str, Any]], prompt: bytes, model: str, effort: str, version: str
) -> tuple[dict[str, Any], bytes]:
    """Bind complete stored originals, preserving their retention limitations."""

    # Require the complete stored sequence, including its original start.
    if (
        not records
        or records[0].get("type") != "session_meta"
        or any(
            type(row.get("ordinal")) is not int or row["ordinal"] != index
            for index, row in enumerate(records)
        )
    ):
        raise CaptureError("original source ordinal continuity unavailable")

    # Reject absent or ambiguous native session/context boundaries.
    metas = [row["payload"] for row in records if row.get("type") == "session_meta"]
    contexts = [
        (ordinal, row["payload"])
        for ordinal, row in enumerate(records, 1)
        if row.get("type") == "turn_context"
    ]
    if len(metas) != 1 or not contexts:
        raise CaptureError("missing or ambiguous native thread/context")

    # Reject inherited or contradictory native ancestry.
    meta = metas[0]
    thread_id = meta.get("id")
    if (
        not isinstance(thread_id, str)
        or not thread_id
        or meta.get("session_id") != thread_id
    ):
        raise CaptureError("missing or conflicting native thread identity")
    if (
        meta.get("parent_thread_id") is not None
        or meta.get("source") != "exec"
        or meta.get("thread_source") != "user"
    ):
        raise CaptureError("native session is not a fresh top-level exec")

    # Bind the installed native version and actual effective turn seat.
    if meta.get("cli_version") != version:
        raise CaptureError("actual native version differs from freeze")
    context_ordinal, context = contexts[0]
    turn_id = context.get("turn_id")
    if (
        not isinstance(turn_id, str)
        or not turn_id
        or context.get("root_turn_id") != turn_id
    ):
        raise CaptureError("missing or conflicting native turn identity")
    if context.get("model") != model or context.get("effort") != effort:
        raise CaptureError("actual native model/effort differs from freeze")

    # Reject contradictory actual seat representations in every context.
    for _, effective in contexts:
        settings = effective.get("collaboration_mode", {}).get("settings", {})
        if settings.get("model") != model or settings.get("reasoning_effort") != effort:
            raise CaptureError("contradictory native context seat")

    # Account for each same-task context through an ordered compaction boundary.
    compactions = [
        (ordinal, row["payload"])
        for ordinal, row in enumerate(records, 1)
        if row.get("type") == "compacted"
    ]
    native_compactions = [
        (ordinal, row["payload"])
        for ordinal, row in enumerate(records, 1)
        if row.get("type") == "event_msg"
        and row.get("payload", {}).get("type") == "item_completed"
        and row["payload"].get("item", {}).get("type") == "ContextCompaction"
    ]
    if len(native_compactions) != len(compactions):
        raise CaptureError("native compaction completion missing/duplicate")
    identities = [payload["item"].get("id") for _, payload in native_compactions]
    if any(
        not isinstance(identity, str) or not identity for identity in identities
    ) or len(set(identities)) != len(identities):
        raise CaptureError("native compaction completion identity unavailable")
    if len(contexts) != len(compactions) + 1:
        raise CaptureError("unexplained native context transition")
    first_window = meta.get("context_window", {}).get("window_id")
    previous_window = first_window
    seen_windows = {first_window}
    for index, (ordinal, later) in enumerate(contexts[1:]):
        if any(
            later.get(key) != context.get(key)
            for key in (
                "turn_id",
                "root_turn_id",
                "model",
                "effort",
                "multi_agent_version",
                "comp_hash",
            )
        ):
            raise CaptureError("compaction changed native task or seat")
        if not contexts[index][0] < compactions[index][0] < ordinal:
            raise CaptureError("native compaction/context order disagrees")
        if not compactions[index][0] < ordinal < native_compactions[index][0]:
            raise CaptureError("native compaction completion order disagrees")
        compacted = compactions[index][1]
        window = compacted.get("window_id")
        if (
            not isinstance(first_window, str)
            or not first_window
            or not isinstance(window, str)
            or not window
            or window in seen_windows
            or compacted.get("first_window_id") != first_window
            or compacted.get("previous_window_id") != previous_window
            or type(compacted.get("window_number")) is not int
            or compacted.get("window_number") != index + 1
        ):
            raise CaptureError("native compaction window chain disagrees")
        seen_windows.add(window)
        previous_window = window

        # Bind the producer's compaction response and resume task to this seat.
        usage = compacted.get("latest_token_usage_record", {})
        resume = compacted.get("resume_metadata", {})
        response_id = compacted.get("compaction_response_id")
        history = compacted.get("replacement_history")
        history_metadata = compacted.get("replacement_history_metadata")
        retained_context = compacted.get("retained_context", {})
        if (
            not isinstance(response_id, str)
            or not response_id
            or any(
                usage.get(key) != value
                for key, value in {
                    "thread_id": thread_id,
                    "session_id": thread_id,
                    "turn_id": turn_id,
                    "root_turn_id": turn_id,
                    "response_id": response_id,
                }.items()
            )
            or resume.get("last_started_turn_id") != turn_id
            or resume.get("multi_agent_version") != context.get("multi_agent_version")
            or any(
                resume.get("previous_turn_settings", {}).get(key) != context.get(key)
                for key in ("model", "comp_hash")
            )
            or not isinstance(compacted.get("message"), str)
            or not isinstance(history, list)
            or not history
            or not isinstance(history_metadata, list)
            or len(history_metadata) != len(history)
            or any(
                retained_context.get(key) is not False
                for key in (
                    "incomplete",
                    "user_messages_incomplete",
                    "assistant_messages_incomplete",
                )
            )
        ):
            raise CaptureError("incomplete or foreign native compaction source")

    # Select the initial task boundary, never a later matching user echo.
    inputs: list[tuple[int, dict[str, Any]]] = []
    finals: list[tuple[int, dict[str, Any]]] = []
    native_inputs: list[tuple[int, dict[str, Any]]] = []
    native_finals: list[tuple[int, dict[str, Any]]] = []
    starts: list[tuple[int, dict[str, Any]]] = []
    completions: list[tuple[int, dict[str, Any]]] = []
    for ordinal, row in enumerate(records, 1):
        # Collect retained message boundaries, including marked earlier tasks.
        payload = row.get("payload", {})
        if row.get("type") == "event_msg" and any(
            key in payload and payload[key] != expected
            for key, expected in {
                "thread_id": thread_id,
                "turn_id": turn_id,
                "root_turn_id": turn_id,
            }.items()
        ):
            raise CaptureError("foreign native event identity")
        if row.get("type") == "response_item" and payload.get("type") == "message":
            kinds = payload.get("internal_chat_message_metadata_passthrough", {}).get(
                "content_item_kinds", []
            )
            if payload.get("role") == "user" and (
                ordinal > context_ordinal
                or "user_input_order" in row.get("metadata", {})
                or "user.text" in kinds
            ):
                inputs.append((ordinal, row))
            if (
                payload.get("role") == "assistant"
                and payload.get("phase") == "final_answer"
            ):
                finals.append((ordinal, row))

        # Collect native lifecycle and completed boundaries with their identity.
        if row.get("type") == "event_msg":
            if payload.get("type") == "task_started":
                starts.append((ordinal, payload))
            if payload.get("type") == "task_complete":
                completions.append((ordinal, payload))

            # Reject foreign completed items before retaining their boundary.
            if payload.get("type") == "item_completed":
                if (
                    payload.get("thread_id") != thread_id
                    or payload.get("turn_id") != turn_id
                ):
                    raise CaptureError("native completed item identity disagrees")
                item = payload.get("item", {})
                if item.get("type") == "UserMessage":
                    native_inputs.append((ordinal, payload))
                if (
                    item.get("type") == "AgentMessage"
                    and item.get("phase") == "final_answer"
                ):
                    native_finals.append((ordinal, payload))

    # Reject incomplete or duplicate representations before pairing them.
    if any(
        len(items) != 1
        for items in (inputs, finals, native_inputs, native_finals, starts, completions)
    ):
        raise CaptureError("missing or ambiguous initial input/final/native completion")

    # Pair the unique retained and independently completed boundaries.
    input_ordinal, input_row = inputs[0]
    native_input_ordinal, native_input = native_inputs[0]
    final_ordinal, final_row = finals[0]
    native_final_ordinal, native_final = native_finals[0]
    start_ordinal, start = starts[0]
    completion_ordinal, completion = completions[0]

    # Reject altered or displaced initial user input.
    if (
        type(input_row.get("metadata", {}).get("user_input_order")) is not int
        or input_row["metadata"]["user_input_order"] != 0
        or input_row["payload"]
        .get("internal_chat_message_metadata_passthrough", {})
        .get("content_item_kinds")
        != ["user.text"]
        or original_message(input_row, turn_id, "user") != prompt
        or native_item_bytes(native_input, thread_id, turn_id) != prompt
    ):
        raise CaptureError("authentic initial user content or order disagrees")

    # Reject conflicting final bytes, identity or completed-turn evidence.
    result = original_message(final_row, turn_id, "assistant")
    if (
        native_final["item"].get("id") != final_row["payload"]["id"]
        or native_item_bytes(native_final, thread_id, turn_id) != result
        or completion.get("turn_id") != turn_id
        or completion.get("last_agent_message", "").encode("utf-8") != result
        or start.get("turn_id") != turn_id
        or start.get("root_turn_id") != turn_id
    ):
        raise CaptureError("native terminal content or turn identity disagrees")

    # Reject reordered boundaries that cannot bind the initial task to output.
    if (
        not start_ordinal
        < context_ordinal
        < input_ordinal
        < native_input_ordinal
        < native_final_ordinal
        < final_ordinal
        < completion_ordinal
    ):
        raise CaptureError("native record order does not bind initial input to result")

    # Every compaction and renewed context stays inside the original task.
    if any(
        not native_input_ordinal < ordinal < native_final_ordinal
        for ordinal, _ in compactions
    ) or any(
        not start_ordinal < ordinal < native_final_ordinal for ordinal, _ in contexts
    ):
        raise CaptureError("native context/compaction escaped original task")

    # CLI0.160.0 completes the native compaction item after the renewed context.
    for index, (ordinal, _) in enumerate(native_compactions):
        following = (
            compactions[index + 1][0]
            if index + 1 < len(compactions)
            else native_final_ordinal
        )
        if ordinal >= following:
            raise CaptureError("native compaction completion escaped its window")

    # Corroborate each replacement against originals already stored before it.
    for ordinal, compacted in compactions:
        corroborate_compaction(records[: ordinal - 1], compacted, input_row, prompt)

    # Return verifiable identities and exact byte digests.
    return {
        "source_contract": "original-completed-v1",
        "initial_retained_source": input_row["metadata"]["retained_source"],
        "final_retained_source": final_row["metadata"]["retained_source"],
        "native_input_id": native_input["item"]["id"],
        "original_input_id": input_row["payload"]["id"],
        "final_id": final_row["payload"]["id"],
        "thread_id": thread_id,
        "turn_id": turn_id,
        "input_ordinal": input_ordinal,
        "native_input_ordinal": native_input_ordinal,
        "final_ordinal": final_ordinal,
        "native_final_ordinal": native_final_ordinal,
        "completion_ordinal": completion_ordinal,
        "input_sha256": digest(prompt),
        "result_sha256": digest(result),
        "model": model,
        "effort": effort,
        "cli_version": version,
    }, result


def bind_exec(
    events: list[dict[str, Any]], thread_id: str, result: bytes,
) -> dict[str, Any]:
    """Normalize malformed exec source without accepting another contract."""

    # Treat unsupported decoded shapes as unavailable original transport.
    try:
        return _bind_exec(events, thread_id, result)
    except (KeyError, TypeError, AttributeError, ValueError) as exc:
        raise CaptureError("malformed exec source representation") from exc


def _bind_exec(
    events: list[dict[str, Any]],
    thread_id: str,
    result: bytes,
) -> dict[str, Any]:
    """Bind exec's own identities and final bytes without equating ID namespaces."""

    # Require one successful ordered exec thread and turn without failure.
    boundaries = {
        kind: [
            (ordinal, row)
            for ordinal, row in enumerate(events, 1)
            if row.get("type") == kind
        ]
        for kind in ("thread.started", "turn.started", "turn.completed")
    }
    if any(len(rows) != 1 for rows in boundaries.values()) or any(
        row.get("type") in ("error", "turn.failed") for row in events
    ):
        raise CaptureError("missing, duplicate or failed exec boundary")
    thread_ordinal, thread = boundaries["thread.started"][0]
    start_ordinal, _ = boundaries["turn.started"][0]
    completion_ordinal, _ = boundaries["turn.completed"][0]
    completed = [
        (ordinal, row["item"])
        for ordinal, row in enumerate(events, 1)
        if row.get("type") == "item.completed"
    ]
    identities = [item.get("id") for _, item in completed]
    finals = [
        (ordinal, item)
        for ordinal, item in completed
        if item.get("type") == "agent_message"
    ]
    if (
        thread.get("thread_id") != thread_id
        or any(not isinstance(identity, str) or not identity for identity in identities)
        or len(set(identities)) != len(identities)
        or not finals
    ):
        raise CaptureError("conflicting exec message or thread identity")

    # The final completed agent item agrees byte-for-byte with raw originals.
    final_ordinal, final = finals[-1]
    if (
        not thread_ordinal < start_ordinal < final_ordinal < completion_ordinal
        or completion_ordinal != len(events)
        or not isinstance(final.get("text"), str)
        or final["text"].encode("utf-8") != result
    ):
        raise CaptureError("exec final content or boundary order disagrees")

    return {
        "final_id": final["id"],
        "final_ordinal": final_ordinal,
        "completion_ordinal": completion_ordinal,
        "result_sha256": digest(result),
    }


def bind_consumption(
    records: list[dict[str, Any]], binding: dict[str, Any], result: bytes,
    specification: dict[str, Any],
) -> dict[str, Any]:
    """Fail closed on unavailable or malformed parent/child proof sources."""

    # Partial receipt files and unsupported shapes cannot become certificates.
    try:
        return _bind_consumption(records, binding, result, specification)
    except (OSError, KeyError, TypeError, AttributeError, ValueError) as exc:
        raise CaptureError("malformed or unavailable consumption source") from exc


def _bind_consumption(
    records: list[dict[str, Any]],
    binding: dict[str, Any],
    result: bytes,
    specification: dict[str, Any],
) -> dict[str, Any]:
    """Bind a separately acquired child's whole return to actual parent input."""

    # Reauthenticate the child packet from its complete original sources.
    child = Path(specification["packet"])
    receipt = json.loads((child / "receipt.json").read_bytes())
    request = json.loads((child / "request.json").read_bytes())
    cleanup = json.loads((child / "cleanup.json").read_bytes())
    if (
        receipt.get("status") != "complete"
        or receipt.get("exit") != 0
        or request.get("source_contract") != "original-completed-v1"
        or cleanup.get("exists") is not False
        or cleanup.get("owned_group_ended") is not True
    ):
        raise CaptureError("consumption child acquisition or cleanup incomplete")
    if len(list((child / "native-sessions").rglob("*.jsonl"))) != 1:
        raise CaptureError("consumption child has extra or missing sessions")
    prelaunch = request.get("prelaunch_sources", {})
    if prelaunch.get("helper_sha256") != digest(
        Path(__file__).read_bytes()
    ) or prelaunch.get("recorder_sha256") != prelaunch.get("helper_sha256"):
        raise CaptureError("consumption child helper identity conflicts")
    sources = receipt["binding"]["sources"]
    captured: dict[str, bytes] = {}
    for kind in ("raw", "events", "prompt", "terminal"):
        path = Path(sources[kind]["path"])
        if path.is_absolute() or ".." in path.parts:
            raise CaptureError("consumption child source path conflicts")
        captured[kind] = (child / path).read_bytes()
        if (
            digest(captured[kind]) != sources[kind]["sha256"]
            or len(captured[kind]) != sources[kind]["bytes"]
        ):
            raise CaptureError("consumption child original source changed")
    child_binding, child_result = bind_original_trace(
        original_jsonl(captured["raw"]),
        captured["prompt"],
        request["model"],
        request["effort"],
        request["version"],
    )
    exec_binding = bind_exec(
        original_jsonl(captured["events"]), child_binding["thread_id"], child_result
    )
    if (
        request.get("prompt_sha256") != digest(captured["prompt"])
        or request.get("prompt_bytes") != len(captured["prompt"])
        or any(
            receipt["binding"].get(key) != value for key, value in child_binding.items()
        )
        or receipt["binding"].get("exec") != exec_binding
    ):
        raise CaptureError("consumption child request or original binding conflicts")
    if (
        child_binding["thread_id"] == binding["thread_id"]
        or captured["terminal"] != child_result
        or (child / "result.txt").read_bytes() != child_result
        or child_result != result
    ):
        raise CaptureError("consumption parent/child identity or final bytes conflict")

    # Pair the unique declared tool call, its command and its model-facing output.
    calls = [
        (ordinal, row["payload"])
        for ordinal, row in enumerate(records, 1)
        if row.get("type") == "response_item"
        and row["payload"].get("type") == "custom_tool_call"
        and row["payload"].get("input") == specification["tool_input"]
    ]
    commands = [
        (ordinal, row["payload"]["item"])
        for ordinal, row in enumerate(records, 1)
        if row.get("type") == "event_msg"
        and row["payload"].get("type") == "item_completed"
        and row["payload"].get("item", {}).get("type") == "CommandExecution"
        and row["payload"]["item"].get("command") == specification["command"]
    ]
    if len(calls) != 1 or len(commands) != 1:
        raise CaptureError("consumption command or call missing/duplicate")
    call_ordinal, call = calls[0]
    command_ordinal, command = commands[0]
    outputs = [
        (ordinal, row["payload"])
        for ordinal, row in enumerate(records, 1)
        if row.get("type") == "response_item"
        and row["payload"].get("type") == "custom_tool_call_output"
        and row["payload"].get("call_id") == call.get("call_id")
    ]
    if len(outputs) != 1:
        raise CaptureError("consumption tool output missing/duplicate")
    output_ordinal, output = outputs[0]
    actual_input = message_bytes({"content": output.get("output")})
    if (
        not binding["native_input_ordinal"]
        < call_ordinal
        < command_ordinal
        < output_ordinal
        < binding["native_final_ordinal"]
        or call.get("name") != "exec"
        or any(
            item.get("internal_chat_message_metadata_passthrough", {}).get("turn_id")
            != binding["turn_id"]
            for item in (call, output)
        )
        or command.get("status") != "completed"
        or command.get("exit_code") != 0
        or command.get("stdout", "").encode("utf-8") != child_result
        or child_result not in command.get("formatted_output", "").encode("utf-8")
        or child_result not in actual_input
    ):
        raise CaptureError("consumption was incomplete or out of original order")

    # The same parent actually executed its separately declared child launcher.
    launches = [
        (ordinal, row["payload"])
        for ordinal, row in enumerate(records, 1)
        if row.get("type") == "response_item"
        and row["payload"].get("type") == "custom_tool_call"
        and row["payload"].get("input") == specification["launch_tool_input"]
    ]
    launch_commands = [
        (ordinal, row["payload"]["item"])
        for ordinal, row in enumerate(records, 1)
        if row.get("type") == "event_msg"
        and row["payload"].get("type") == "item_completed"
        and row["payload"].get("item", {}).get("type") == "CommandExecution"
        and row["payload"]["item"].get("command") == specification["launch_command"]
    ]
    if len(launches) != 1 or len(launch_commands) != 1:
        raise CaptureError("consumption child launcher source missing/duplicate")
    launch_ordinal, launch = launches[0]
    launch_command_ordinal, launch_command = launch_commands[0]
    if (
        not binding["native_input_ordinal"]
        < launch_ordinal
        < launch_command_ordinal
        < call_ordinal
        or launch.get("name") != "exec"
        or launch.get("internal_chat_message_metadata_passthrough", {}).get("turn_id")
        != binding["turn_id"]
        or launch_command.get("status") != "completed"
        or launch_command.get("exit_code") != 0
    ):
        raise CaptureError("consumption child launcher identity/order conflicts")

    return {
        "launch_command": launch_command["command"],
        "launch_command_id": launch_command["id"],
        "launch_ordinal": launch_ordinal,
        "child_thread_id": child_binding["thread_id"],
        "child_turn_id": child_binding["turn_id"],
        "call_id": call["call_id"],
        "command_id": command["id"],
        "call_ordinal": call_ordinal,
        "command_ordinal": command_ordinal,
        "output_ordinal": output_ordinal,
        "command": command["command"],
        "cwd": command.get("cwd"),
        "result_bytes": len(result),
        "result_sha256": digest(result),
        "parent_input_sha256": digest(actual_input),
    }


def original_jsonl(data: bytes) -> list[dict[str, Any]]:
    """Decode stored source while rejecting duplicate or partial JSON fields."""

    # A later duplicate key cannot silently replace a contradictory original.
    def object_fields(pairs: list[tuple[str, Any]]) -> dict[str, Any]:
        """Preserve each unique decoded object field."""
        if len({key for key, _ in pairs}) != len(pairs):
            raise CaptureError("conflicting duplicate original JSON field")
        return dict(pairs)

    # Malformed or non-object rows cannot establish complete source boundaries.
    try:
        rows = [
            json.loads(line, object_pairs_hook=object_fields)
            for line in data.decode("utf-8").splitlines()
        ]
    except (UnicodeError, ValueError) as exc:
        raise CaptureError("partial or malformed original JSON source") from exc
    if not rows or any(not isinstance(row, dict) for row in rows):
        raise CaptureError("original JSON record shape unavailable")

    return rows


def source_observation(
    records: list[dict[str, Any]],
    raw: bytes,
    paths: dict[str, Path],
    packet: Path,
) -> dict[str, Any]:
    """Retain byte identities, offsets and separate compaction retention facts."""

    # Index original record bytes rather than a later JSON serialization.
    index: list[dict[str, Any]] = []
    offset = 0
    for row, line in zip(records, raw.splitlines(keepends=True), strict=True):
        index.append(
            {
                "ordinal": row["ordinal"],
                "offset": offset,
                "bytes": len(line),
                "sha256": digest(line),
            }
        )
        offset += len(line)

    # Preserve observed window and retention metadata with opaque text digests.
    compactions = []
    for row in records:
        if row.get("type") != "compacted":
            continue
        payload = row["payload"]
        retained = payload["retained_context"]
        observation = {
            key: value
            for key, value in retained.items()
            if key not in ("user_messages", "assistant_messages")
        }
        for role in ("user", "assistant"):
            observation[f"{role}_messages"] = [
                {
                    **{key: value for key, value in message.items() if key != "text"},
                    "utf8_bytes": len(message["text"].encode("utf-8")),
                    "sha256": digest(message["text"].encode("utf-8")),
                }
                for message in retained[f"{role}_messages"]
            ]
        compactions.append(
            {
                "ordinal": row["ordinal"],
                **{
                    key: payload[key]
                    for key in (
                        "window_number",
                        "first_window_id",
                        "previous_window_id",
                        "window_id",
                        "compaction_response_id",
                    )
                },
                "retained_context": observation,
            }
        )

    # Name the permanent copied rollout; other sources are already permanent.
    sources = {}
    for kind, path in paths.items():
        data = raw if kind == "raw" else path.read_bytes()
        destination = (
            Path("native-sessions")
            / path.relative_to(packet / "runtime/home/.codex/sessions")
            if kind == "raw"
            else path.relative_to(packet)
        )
        sources[kind] = {
            "path": str(destination),
            "bytes": len(data),
            "sha256": digest(data),
        }

    return {"sources": sources, "record_index": index, "compactions": compactions}


def inventory(
    roots: list[Path], credential_paths: list[Path] | None = None
) -> dict[str, Any]:
    """Track paths, kinds, modes and bytes; redact only enumerated credentials."""

    # Exact private paths protect credentials without hiding ordinary auth.json.
    credentials = {path.absolute() for path in credential_paths or []}
    result: dict[str, Any] = {}

    # Inventory each root and descendant without following symlink targets.
    for root in roots:
        for path in [root, *sorted(root.rglob("*"))]:
            metadata = path.lstat()
            entry: dict[str, Any] = {"mode": stat.S_IMODE(metadata.st_mode)}
            if stat.S_ISLNK(metadata.st_mode):
                entry.update(kind="symlink", symlink=os.readlink(path))
            elif stat.S_ISDIR(metadata.st_mode):
                entry["kind"] = "directory"
            elif stat.S_ISREG(metadata.st_mode):
                redacted = path.absolute() in credentials
                entry.update(
                    kind="file",
                    bytes=metadata.st_size,
                    sha256=None if redacted else digest(path.read_bytes()),
                )
                if redacted:
                    entry["credential_redacted"] = True
            else:
                entry.update(kind="other", file_type=stat.S_IFMT(metadata.st_mode))
            result[str(path)] = entry

    return result


def environment(packet: Path) -> dict[str, str]:
    """Keep acquisition and lifecycle tool storage inside the owned packet."""

    # Keep lifecycle receipts local and suppress recorder bytecode.
    env = dict(os.environ)
    runtime = packet / "runtime"
    env.update(KNTNT_HOME=str(packet / "lifecycle"), PYTHONDONTWRITEBYTECODE="1")

    # Scope auxiliary tool storage to the owned runtime.
    for name in (
        "TMPDIR",
        "UV_CACHE_DIR",
        "UV_TOOL_DIR",
        "UV_TOOL_BIN_DIR",
        "UV_PYTHON_INSTALL_DIR",
        "UV_PROJECT_ENVIRONMENT",
        "XDG_CACHE_HOME",
        "XDG_DATA_HOME",
        "XDG_CONFIG_HOME",
    ):
        env[name] = str(runtime / name.lower())

    return env


def process_start(pid: int, cwd: Path) -> str:
    """Read the same platform start identity used by installed session cleanup."""

    # An unreadable or vanished start never authorizes a prospective signal.
    try:
        observed = subprocess.run(
            ["ps", "-o", "lstart=", "-p", str(pid)],
            cwd=cwd,
            capture_output=True,
            text=True,
            check=False,
            timeout=5,
        )
    except (OSError, subprocess.SubprocessError) as exc:
        raise CaptureError("native ownership start identity unreadable") from exc
    return observed.stdout.strip()


def stop_owned(
    child: subprocess.Popen[bytes],
    expected_start: str | None = None,
    cwd: Path | None = None,
) -> None:
    """End an owned process group even when its leader already exited."""

    # Escalate termination only within the known owned group.
    for sig in (signal.SIGTERM, signal.SIGKILL):
        # Recheck a live leader before each signal; reused groups stay untouched.
        if expected_start is not None:
            if not expected_start or cwd is None:
                raise CaptureError("native ownership start identity unavailable")
            try:
                group = os.getpgid(child.pid)
            except ProcessLookupError:
                group = None
            if group is not None and (
                group != child.pid or process_start(child.pid, cwd) != expected_start
            ):
                raise CaptureError("native ownership identity changed")

        # Stop if the owned group has already vanished.
        try:
            os.killpg(child.pid, sig)
        except ProcessLookupError:
            return
        except OSError as exc:
            if expected_start is not None:
                raise CaptureError("native ownership signal unavailable") from exc
            raise

        # Bound the leader wait before escalating unresolved ownership.
        try:
            child.wait(timeout=5)
        except subprocess.TimeoutExpired:
            continue

        # Check surviving descendants after the group leader exits.
        try:
            os.killpg(child.pid, 0)
        except ProcessLookupError:
            return

    # Report unresolved ownership while its runtime is still reachable.
    raise CaptureError("owned native process group remains after termination")


def register(pid: int, cleanup: Path, env: dict[str, str], packet: Path) -> None:
    """Register each group before supplying a native task."""

    # Register the original group through the installed lifecycle contract.
    receipt = subprocess.run(
        [
            "uv",
            "run",
            str(cleanup),
            "add",
            "pid",
            str(pid),
            "Prospective native evaluation dispatch",
        ],
        env=env,
        cwd=packet,
        capture_output=True,
        text=True,
        check=True,
        timeout=30,
    )

    # Preserve the issued registration and actual observed process group.
    with (packet / "registrations.jsonl").open("a") as stream:
        stream.write(
            json.dumps({"pid": pid, "pgid": os.getpgid(pid), "receipt": receipt.stdout})
            + "\n"
        )


def acquire(args: argparse.Namespace, prompt: bytes) -> bytes:
    """Capture one attempt, retaining unsuccessful receipts without retrying."""

    source_contract = getattr(args, "source_contract", "retained-source-v1")
    if source_contract not in ("retained-source-v1", "original-completed-v1"):
        raise CaptureError("unknown source contract")
    consumption = None
    consume_spec = getattr(args, "consume_spec", None)
    if consume_spec is not None:
        if source_contract != "original-completed-v1":
            raise CaptureError("consumption requires the explicit original contract")
        data = consume_spec.read_bytes()
        consumption = original_jsonl(data)[0]
        if (
            not isinstance(consumption.get("packet"), str)
            or not isinstance(consumption.get("command"), list)
            or not consumption["command"]
            or any(not isinstance(word, str) for word in consumption["command"])
            or not isinstance(consumption.get("tool_input"), str)
            or not consumption["tool_input"]
        ):
            raise CaptureError("consumption specification unavailable")
        (args.packet / "consumption-spec.json").write_bytes(data)
    compact_limit = getattr(args, "auto_compact_token_limit", None)
    if compact_limit is not None and (
        source_contract != "original-completed-v1"
        or type(compact_limit) is not int
        or compact_limit <= 0
    ):
        raise CaptureError(
            "compaction instrumentation requires explicit original contract and a positive integer"
        )
    if source_contract == "original-completed-v1" and (
        not math.isfinite(args.timeout) or not 0 < args.timeout <= 1200
    ):
        raise CaptureError("native deadline must be finite and at most 1200 seconds")

    # Reject unsupported dispatch bytes before invoking the native CLI.
    prompt.decode("utf-8")
    if not prompt:
        raise CaptureError("empty dispatch brief")

    # Allocate private native state underneath registered task ownership.
    packet: Path = args.packet.resolve()
    runtime = packet / "runtime"
    native_home = runtime / "home" / ".codex"
    native_home.mkdir(parents=True)

    # Check the frozen CLI release with a bounded owned subprocess.
    native = str(args.native.resolve())
    actual_version = subprocess.check_output(
        [native, "--version"],
        cwd=args.cwd,
        env=environment(packet),
        text=True,
        timeout=15,
    ).strip()
    if actual_version != f"codex-cli {args.version}":
        raise CaptureError("installed CLI differs from frozen version")

    # Retain the complete dispatch request independently of native output.
    write_json(
        packet / "request.json",
        {
            "method": "instrumented-native-stdin",
            "model": args.model,
            "effort": args.effort,
            "version": args.version,
            "prompt_sha256": digest(prompt),
            "prompt_bytes": len(prompt),
            "parent": args.parent,
            "role": args.role,
            "cwd": str(args.cwd.resolve()),
            "timeout_seconds": args.timeout,
            **(
                {
                    "source_contract": source_contract,
                    "compaction_instrumentation": None
                    if compact_limit is None
                    else {
                        "model_auto_compact_token_limit": compact_limit,
                        "model_auto_compact_token_limit_scope": "total",
                    },
                    "prelaunch_sources": {
                        "helper_path": str(Path(__file__).resolve()),
                        "helper_sha256": digest(Path(__file__).read_bytes()),
                        "recorder_sha256": digest(Path(__file__).read_bytes()),
                        "native_path": native,
                        "native_sha256": digest(Path(native).read_bytes()),
                    },
                }
                if source_contract == "original-completed-v1"
                else {}
            ),
        },
    )

    # Copy active lifecycle hooks and native configuration into private state.
    hooks = args.auth_home / "hooks.json"
    if not hooks.is_file():
        raise CaptureError("existing lifecycle hook configuration unavailable")
    shutil.copy2(hooks, native_home / "hooks.json")
    config = args.auth_home / "config.toml"
    if config.is_file():
        shutil.copy2(config, native_home / "config.toml")

    # Record copied configuration identities and enabled hook execution.
    write_json(
        packet / "configuration.json",
        {
            "hooks_sha256": digest(hooks.read_bytes()),
            "config_sha256": digest(config.read_bytes()) if config.is_file() else None,
            "hooks_enabled_by_command": True,
        },
    )

    # Protect the private credential copy with restrictive permissions.
    shutil.copy2(args.auth_home / "auth.json", native_home / "auth.json")
    (native_home / "auth.json").chmod(0o600)

    # Build private HOME and auxiliary storage from the owned environment.
    env = environment(packet)
    env.update(CODEX_HOME=str(native_home))
    if source_contract == "retained-source-v1":
        env["HOME"] = str(runtime / "home")
    for name in (
        "TMPDIR",
        "UV_CACHE_DIR",
        "UV_TOOL_DIR",
        "UV_TOOL_BIN_DIR",
        "UV_PYTHON_INSTALL_DIR",
        "UV_PROJECT_ENVIRONMENT",
        "XDG_CACHE_HOME",
        "XDG_DATA_HOME",
        "XDG_CONFIG_HOME",
    ):
        Path(env[name]).mkdir(exist_ok=True)

    # Inventory declared writable roots with one exact credential exclusion.
    roots = [args.cwd.resolve(), packet, *[p.resolve() for p in args.inventory_root]]
    credentials = [native_home / "auth.json"]
    write_json(packet / "inventory-before.json", inventory(roots, credentials))

    # Declare one native attempt and its initially unsuccessful receipt.
    command = [
        native,
        "exec",
        "-C",
        str(args.cwd.resolve()),
        "--skip-git-repo-check",
        "--dangerously-bypass-approvals-and-sandbox",
        "--dangerously-bypass-hook-trust",
        "--enable",
        "hooks",
        "--json",
        "--output-last-message",
        str(packet / "terminal.txt"),
        "-m",
        args.model,
        "-c",
        f'model_reasoning_effort="{args.effort}"',
        "-",
    ]
    if compact_limit is not None:
        command[-1:-1] = [
            "-c",
            f"model_auto_compact_token_limit={compact_limit}",
            "-c",
            'model_auto_compact_token_limit_scope="total"',
        ]
    child: subprocess.Popen[bytes] | None = None
    ownership: dict[str, Any] = {}
    started = time.monotonic()
    receipt: dict[str, Any] = {
        "status": "failed",
        "command": command,
        "runtime": str(runtime),
    }

    try:
        # Register the new native group before sending its complete input.
        with (
            (packet / "events.jsonl").open("wb") as stdout,
            (packet / "stderr.txt").open("wb") as stderr,
        ):
            child = subprocess.Popen(
                command,
                env=env,
                cwd=args.cwd,
                stdin=subprocess.PIPE,
                stdout=stdout,
                stderr=stderr,
                start_new_session=True,
            )
            receipt.update({"pid": child.pid, "pgid": child.pid})
            register(child.pid, args.cleanup_script, env, packet)
            if source_contract == "original-completed-v1":
                ownership = {
                    "pid": child.pid,
                    "pgid": os.getpgid(child.pid),
                    "started": process_start(child.pid, packet),
                }
                receipt["ownership"] = ownership
                if ownership["pgid"] != child.pid or not ownership["started"]:
                    raise CaptureError("native ownership identity unavailable")
            remaining = args.timeout
            if source_contract == "original-completed-v1":
                remaining -= time.monotonic() - started
                if remaining <= 0:
                    raise CaptureError("native deadline expired before supplying input")
            child.communicate(prompt, timeout=remaining)

        # Reject unsuccessful or ambiguous native completion.
        receipt["exit"] = child.returncode
        if child.returncode:
            raise CaptureError("native child did not complete successfully")
        sessions = sorted((native_home / "sessions").rglob("*.jsonl"))
        if len(sessions) != 1:
            raise CaptureError("missing or extra native sessions")

        # Bind raw input/context and terminal bytes to the completed result.
        records = (
            original_jsonl(sessions[0].read_bytes())
            if source_contract == "original-completed-v1"
            else [json.loads(line) for line in sessions[0].read_text().splitlines()]
        )
        if source_contract == "original-completed-v1":
            binder = bind_original_trace
        elif source_contract == "retained-source-v1":
            binder = bind_trace
        else:
            raise CaptureError("unknown source contract")
        binding, result = binder(records, prompt, args.model, args.effort, args.version)
        if source_contract == "original-completed-v1":
            events = original_jsonl((packet / "events.jsonl").read_bytes())
            binding["exec"] = bind_exec(events, binding["thread_id"], result)
        terminal = (packet / "terminal.txt").read_bytes()
        if terminal != result:
            raise CaptureError("CLI terminal bytes differ from native final message")

        if source_contract == "original-completed-v1":
            binding.update(
                source_observation(
                    records,
                    sessions[0].read_bytes(),
                    {
                        "raw": sessions[0],
                        "events": packet / "events.jsonl",
                        "prompt": packet / "prompt.txt",
                        "terminal": packet / "terminal.txt",
                    },
                    packet,
                )
            )

        if consumption is not None:
            binding["consumption"] = bind_consumption(
                records, binding, result, consumption
            )

        # Retain a complete result only after all native bindings hold.
        receipt.update(
            {
                "status": "bound"
                if source_contract == "original-completed-v1"
                else "complete",
                "binding": binding,
            }
        )
        if source_contract == "retained-source-v1":
            (packet / "result.txt").write_bytes(result)

        return result

    # Preserve native capture or timeout failure without fallback.
    except (CaptureError, subprocess.TimeoutExpired) as exc:
        receipt["reason"] = str(exc)
        raise CaptureError(str(exc)) from exc
    finally:
        # Resolve group ownership before copying or removing private state.
        if child is not None:
            try:
                if source_contract == "original-completed-v1":
                    stop_owned(child, ownership.get("started", ""), packet)
                else:
                    stop_owned(child)
            except CaptureError as exc:
                write_json(
                    packet / "cleanup-failed.json",
                    {"pgid": child.pid, "runtime": str(runtime)},
                )
                if source_contract == "original-completed-v1":
                    receipt.update(
                        status="failed",
                        reason=str(exc),
                        elapsed_seconds=time.monotonic() - started,
                    )
                    write_json(packet / "receipt.json", receipt)
                raise

        # Preserve raw native sessions and timing after group termination.
        receipt["elapsed_seconds"] = time.monotonic() - started
        sessions_root = native_home / "sessions"
        if sessions_root.exists():
            shutil.copytree(sessions_root, packet / "native-sessions")

        # Record the full after-inventory and final acquisition disposition.
        write_json(packet / "inventory-after.json", inventory(roots, credentials))
        write_json(packet / "receipt.json", receipt)


def run(args: argparse.Namespace, prompt: bytes | None = None) -> bytes:
    """Clean private credentials on preparation failures as well as completion."""

    # Allocate a new evidence packet without overwriting an earlier attempt.
    packet = args.packet.resolve()
    packet.mkdir(parents=True, exist_ok=False)
    runtime = packet / "runtime"

    try:
        # Register ownership before stdin or native preparation can block.
        if os.getpid() != os.getpgrp():
            os.setsid()
        register(os.getpid(), args.cleanup_script, environment(packet), packet)

        # Capture supplied stdin unchanged before native preparation begins.
        if prompt is None:
            prompt = sys.stdin.buffer.read()
        (packet / "prompt.txt").write_bytes(prompt)

        # Acquire the single native child after ownership and input capture.
        result = acquire(args, prompt)

    # Catch all preparation errors so unexpected failures retain their receipt.
    except Exception as exc:
        if not (packet / "receipt.json").exists():
            write_json(
                packet / "receipt.json", {"status": "failed", "reason": str(exc)}
            )
        raise
    finally:
        # Remove credentials only after native ownership has been resolved.
        if runtime.exists() and not (packet / "cleanup-failed.json").exists():
            print(f"Delete owned disposable runtime: {runtime}", file=sys.stderr)
            try:
                shutil.rmtree(runtime)
            except OSError as exc:
                if (
                    getattr(args, "source_contract", "retained-source-v1")
                    == "original-completed-v1"
                ):
                    receipt_path = packet / "receipt.json"
                    receipt = json.loads(receipt_path.read_text())
                    receipt.update(status="failed", reason=str(exc))
                    write_json(receipt_path, receipt)
                raise
            write_json(
                packet / "cleanup.json",
                {
                    "deleted_literal_path": str(runtime),
                    "exists": runtime.exists(),
                    "owned_group_ended": True,
                },
            )
    # Publish only after the selected original contract's owned cleanup holds.
    if (
        getattr(args, "source_contract", "retained-source-v1")
        == "original-completed-v1"
    ):
        receipt_path = packet / "receipt.json"
        receipt = json.loads(receipt_path.read_text())
        (packet / "result.txt").write_bytes(result)
        receipt["status"] = "complete"
        write_json(receipt_path, receipt)

    return result


def main() -> int:
    """Print only the complete child return on successful acquisition."""

    # Parse declared acquisition boundaries and finite execution limits.
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--packet", type=Path, required=True)
    parser.add_argument("--cwd", type=Path, required=True)
    parser.add_argument("--model", required=True)
    parser.add_argument("--effort", required=True)
    parser.add_argument("--version", required=True)
    parser.add_argument("--parent", required=True)
    parser.add_argument("--role", required=True)
    parser.add_argument("--timeout", type=float, default=180)
    parser.add_argument(
        "--source-contract",
        choices=("retained-source-v1", "original-completed-v1"),
        default="retained-source-v1",
        help="explicit prospective source claim; no fallback between contracts",
    )
    parser.add_argument(
        "--consume-spec",
        type=Path,
        help="optional frozen neutral parent-consumption specification JSON",
    )
    parser.add_argument(
        "--auto-compact-token-limit",
        type=int,
        help="disclosed total-context instrumentation, original-completed-v1 only",
    )
    parser.add_argument(
        "--native", type=Path, default=Path(shutil.which("codex") or "codex")
    )
    parser.add_argument("--auth-home", type=Path, default=Path.home() / ".codex")
    parser.add_argument(
        "--cleanup-script",
        type=Path,
        default=Path.home()
        / ".agents/skills/kntnt/features/session-cleanup/scripts/session_cleanup.py",
    )
    parser.add_argument("--inventory-root", type=Path, action="append", default=[])
    args = parser.parse_args()

    # Report unavailable acquisitions through failure-only stderr.
    try:
        result = run(args)
    except (CaptureError, OSError, ValueError, subprocess.SubprocessError) as exc:
        print(f"Native acquisition unavailable: {exc}", file=sys.stderr)
        return 1

    # Transport the complete child result without appending bytes.
    sys.stdout.buffer.write(result)

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
