"""Run the four unchanged project checks and retain their complete handoff.

This operator snapshot is not a product implementation. Runtime writes and
cleanup registration are confined to this ticket's resolved assigned scratch.
"""

import json
import os
import shlex
import subprocess
import time
from pathlib import Path

work = Path("/Users/thomas/Projects/skills/.git/kntnt-orchestrate/488")
evidence = work / "docs/evaluation/editorial-488/amend-2"
scratch = Path("/Users/thomas/Projects/skills/.git/kntnt-orchestrate/488.scratch").resolve()
runtime = scratch / "amend-2-runtime"
runtime.mkdir(exist_ok=False)
environment = {
    "TMPDIR": str(runtime / "tmp"),
    "UV_CACHE_DIR": str(runtime / "uv-cache"),
    "UV_TOOL_DIR": str(runtime / "uv-tools"),
    "UV_TOOL_BIN_DIR": str(runtime / "uv-bin"),
    "UV_PYTHON_INSTALL_DIR": str(runtime / "uv-python"),
    "UV_PROJECT_ENVIRONMENT": str(runtime / "project-env"),
    "PYTHONPYCACHEPREFIX": str(runtime / "pycache"),
    "MYPY_CACHE_DIR": str(runtime / "mypy-cache"),
    "RUFF_CACHE_DIR": str(runtime / "ruff-cache"),
    "XDG_CACHE_HOME": str(runtime / "xdg-cache"),
    "XDG_DATA_HOME": str(runtime / "xdg-data"),
    "XDG_RUNTIME_DIR": str(runtime / "xdg-runtime"),
    "KNTNT_HOME": str(runtime / "kntnt-home"),
    "PYTEST_ADDOPTS": f"--basetemp={runtime / 'pytest-temp'} -o cache_dir={runtime / 'pytest-cache'}",
}
for key, value in environment.items():
    if key != "PYTEST_ADDOPTS":
        Path(value).mkdir(parents=True, exist_ok=True)
native_home = os.environ.get("HOME")
native_codex_home = os.environ.get("CODEX_HOME")
os.environ.update(environment)
assert os.environ.get("HOME") == native_home
assert os.environ.get("CODEX_HOME") == native_codex_home
if os.getpid() != os.getpgrp():
    os.setsid()


def write_receipt(name: str, value: object) -> None:
    """Preserve operator observations outside the disposable runtime."""
    (evidence / name).write_text(json.dumps(value, indent=2) + "\n")


def register(kind: str, identifier: str) -> None:
    """Register immediately through the shipped cleanup CLI in this tree."""
    result = subprocess.run(
        [
            "uv", "run",
            str(work / "skills/kntnt/features/session-cleanup/scripts/session_cleanup.py"),
            "add", kind, identifier, "Editorial #488 amend 2 project gate",
        ],
        cwd=work,
        capture_output=True,
        text=True,
        check=True,
    )
    with (evidence / "registration.log").open("a") as log:
        log.write(f"{kind} {identifier}\n{result.stdout}{result.stderr}")


register("path", str(runtime))
register("pid", str(os.getpid()))
commands = [
    "uvx ruff check .",
    "uvx ruff format --check .",
    "uvx --with types-PyYAML --with pytest --with httpx==0.28.1 --with selectolax==0.4.11 --with types-tinycss2 --with types-regex mypy skills/kntnt/scripts/kntnt.py skills/kntnt/library/scripts/argument_grammar.py skills/kntnt/library/scripts/article_anatomy.py skills/kntnt/library/scripts/ship.py skills/kntnt/library/scripts/integrations.py skills/kntnt/library/scripts/languages.py skills/kntnt/library/scripts/credentials.py skills/models/model-selector/scripts/catalogue.py skills/models/model-selector/scripts/profiles.py skills/models/model-selector/scripts/evidence.py skills/models/model-selector/scripts/launch.py skills/models/model-selector/scripts/selection.py skills/models/model-selector/scripts/quota.py skills/models/model-selector/scripts/record.py skills/models/model-selector/scripts/capture.py skills/models/model-selector/scripts/grade.py skills/models/model-selector/scripts/setup_apply.py skills/models/model-selector/scripts/inventory.py skills/code/orchestrate/scripts/run.py skills/productivity/rename-invoices/scripts/rename_invoices.py skills/agents/explain/scripts/invoke.py skills/web/mirror/scripts/mirror.py skills/infrastructure/cloudns/scripts/cloudns.py skills/infrastructure/hetzner/scripts/hetzner.py skills/infrastructure/nodeping/scripts/nodeping.py skills/infrastructure/postmark/scripts/postmark.py tests",
    "uv run --with pytest --with pytest-xdist --with pyyaml pytest -n auto",
]
# Require exact guide commands; neither omissions nor reordered arguments pass.
guide = (work / "CONTRIBUTING.md").read_text()
assert all(f"   {command}\n" in guide for command in commands)
receipt = {
    "starting_head": "9e7817803178831149945ebd8dfb3c1069e0c3cb",
    "supervisor_pid": os.getpid(),
    "supervisor_process_group": os.getpgrp(),
    "started": time.time(),
    "environment": environment,
    "HOME_retained": True,
    "CODEX_HOME_retained": True,
    "commands": [],
}
write_receipt("gate-receipt.json", receipt)
for index, command in enumerate(commands, 1):
    with (evidence / f"gate-{index}.log").open("w") as log:
        log.write(command + "\n")
        log.flush()
        process = subprocess.Popen(
            shlex.split(command),
            cwd=work,
            stdout=log,
            stderr=subprocess.STDOUT,
            start_new_session=True,
        )
        register("pid", str(process.pid))
        item = {
            "command": command,
            "pid": process.pid,
            "process_group": process.pid,
            "started": time.time(),
            "log": f"gate-{index}.log",
        }
        receipt["commands"].append(item)
        write_receipt("gate-receipt.json", receipt)
        item["returncode"] = process.wait()
        item["finished"] = time.time()
        write_receipt("gate-receipt.json", receipt)
    print(f"Gate {index}: exit {item['returncode']}", flush=True)
receipt["finished"] = time.time()
receipt["all_passed"] = all(item["returncode"] == 0 for item in receipt["commands"])
write_receipt("gate-receipt.json", receipt)
raise SystemExit(0 if receipt["all_passed"] else 1)
