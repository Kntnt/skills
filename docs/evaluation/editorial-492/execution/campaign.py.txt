"""Stage and execute the frozen baseline matrix in four sequential input lanes."""
from pathlib import Path
import concurrent.futures
import csv
import hashlib
import json
import os
import secrets
import subprocess
import time
root=Path('/private/tmp/kntnt-orchestrate-20261002.o7bmMK/role-1ec1788d/evaluation')
repo=Path('/Users/thomas/Projects/skills/.git/kntnt-orchestrate/492')
base=repo/'docs/evaluation/editorial-492'
if os.getpgrp()!=os.getpid():
    os.setsid()
subprocess.run(['uv','run',str(repo/'skills/kntnt/features/session-cleanup/scripts/session_cleanup.py'),'add','pid',str(os.getpid()),'Ticket 492 fresh native baseline lanes'],cwd=repo,check=True,capture_output=True)

def inventory(name):
    files={}
    for p in sorted(root.rglob('*')):
        if p.is_file() and not p.is_symlink():
            try:
                s=p.stat()
                files[str(p.relative_to(root))]={'bytes':s.st_size,'mtime_ns':s.st_mtime_ns}
            except FileNotFoundError:
                pass
    status=subprocess.check_output(['git','status','--porcelain','--untracked-files=all'],cwd=repo,text=True)
    head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=repo,text=True).strip()
    (root/f'wave-{name}.json').write_text(json.dumps({'time':time.time(),'head':head,'git_status':status,'scratch':files},indent=2)+'\n')
rows=[r for r in csv.DictReader((base/'matrix.tsv').open(),delimiter='\t') if r['arm']=='pre']
for r in rows:
    r['token']=secrets.token_hex(6)
    fixture=r['fixture']
    rev='4cf05813' if r['source']=='corpus' else 'cb446552'
    path=f'docs/evaluation/corpus/editorial-quality/controls/{fixture}.md' if r['source']=='corpus' else f'docs/evaluation/editorial-492/inputs/{fixture}.md'
    supplied=root/f'{fixture}.md'
    supplied.write_bytes(subprocess.check_output(['git','show',f'{rev}:{path}'],cwd=repo))
    r['input']=str(supplied)
    r['input_revision']=rev
    prompt=root/f'{r["token"]}.prompt.txt'
    prompt.write_text(r['invocation']+'\n')
    r['prompt']=str(prompt)
(root/'runs-map.json').write_text(json.dumps(rows,indent=2)+'\n')
inventory('baseline-before')

def run_lane(fixture):
    for r in rows:
        if r['fixture']!=fixture:
            continue
        cmd=['uv','run',str(base/'harness/run.py'),'--revision=4cf05813',f'--corpus-revision={r["input_revision"]}',f'--prompt={r["prompt"]}',f'--input={r["input"]}','--input-name=input.md',f'--output={root/r["token"]}']
        if r['target']=='file':
            cmd.append('--capture-output-name=output.md')
        result=subprocess.run(cmd,cwd=repo,capture_output=True,text=True)
        record={'run':r['run'],'token':r['token'],'returncode':result.returncode,'stdout':result.stdout,'stderr':result.stderr}
        (root/f'{r["token"]}.launch.json').write_text(json.dumps(record,indent=2)+'\n')
        print(json.dumps(record),flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
    list(pool.map(run_lane,dict.fromkeys(r['fixture'] for r in rows)))
inventory('baseline-after')
