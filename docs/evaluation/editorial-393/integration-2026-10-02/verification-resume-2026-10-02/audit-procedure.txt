def read(p):
    return json.loads(p.read_text())

def sha(b):
    return hashlib.sha256(b).hexdigest()

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT)

def unfold(value, depth=0):
    if depth > 8:
        return []
    if isinstance(value, dict):
        return sum((unfold(v, depth + 1) for v in value.values()), [])
    if isinstance(value, list):
        return sum((unfold(v, depth + 1) for v in value), [])
    if not isinstance(value, str):
        return []
    out = [value]
    for candidate in [value, *value.splitlines()]:
        try:
            decoded = json.loads(candidate)
        except (ValueError, TypeError):
            continue
        if decoded != value:
            out.extend(unfold(decoded, depth + 1))
    return out

def native(directory):
    sessions = []
    for file in sorted((directory / 'native-sessions').rglob('*.jsonl')):
        rows = [json.loads(line) for line in file.read_text().splitlines()]
        metas = [x['payload'] for x in rows if x['type'] == 'session_meta']
        assert len(metas) == 1
        contexts = [x['payload'] for x in rows if x['type'] == 'turn_context']
        assert contexts and all((x['model'], x['effort']) == ('gpt-6.1-sol', 'xhigh') for x in contexts)
        calls = [x['payload'] for x in rows if x['type'] == 'response_item' and x['payload'].get('type') == 'function_call']
        starts = [json.loads(x['arguments']) for x in calls if x.get('name') == 'spawn_agent']
        assert all(x['fork_turns'] == 'none' and 'model' not in x and 'reasoning_effort' not in x for x in starts)
        outputs = []
        for row in rows:
            payload = row['payload']
            if row['type'] == 'response_item' and payload.get('type') in ['function_call_output', 'custom_tool_call_output']:
                outputs.extend(unfold(payload.get('output')))
        events = [x['payload'] for x in rows if x['type'] == 'event_msg']
        sessions.append({'file': str(file.relative_to(P)), 'id': metas[0]['id'], 'source': metas[0]['source'], 'started': any(x['type'] == 'task_started' for x in events), 'completed': any(x['type'] == 'task_complete' for x in events), 'spawn_count': len(starts), 'outputs': outputs, 'finals': [x.get('last_agent_message') for x in events if x['type'] == 'task_complete']})
    return sessions
